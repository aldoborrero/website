{{- /* Share image, 1200×630, drawn at build time with Hugo's image filters: the default palette (params.palette) in light mode, read from tokens.css; the type label in the accent, the title in Newsreader and the name in Hanken Grotesk. The two TTFs come from the nix fonts derivation (assets/fonts/, copied by `just fonts`); without them there is no image and the page falls back to a small card. Context: dict page, title, label. Returns an image resource or nil. */ -}} {{- $out := false -}} {{- $serif := resources.Get "fonts/og-serif.ttf" -}} {{- $sans := resources.Get "fonts/og-sans.ttf" -}} {{- $base := resources.Get "images/og-base.png" -}} {{- if and $serif $sans $base -}} {{- $root := index (findRE (printf `(?ms)^\[data-palette="%s"\] \{.*?\}` site.Params.palette) (resources.Get "css/tokens.css").Content 1) 0 -}} {{- $c := dict -}} {{- range slice "bg" "ink" "muted" "accent" "line" -}} {{- $m := index (findRESubmatch (printf `--%s:\s*(#[0-9a-fA-F]{6})` .) $root 1) 0 -}} {{- $c = merge $c (dict . (index $m 1)) -}} {{- end -}} {{- $title := .title | truncate 96 -}} {{- $bar := $base | images.Filter (images.Padding 0 119 5 0 $c.accent) -}} {{- $filters := slice (images.Padding 0 1199 629 0 $c.bg) (images.Overlay $bar 80 96) (images.Text $title (dict "font" $serif "color" $c.ink "size" 68 "linespacing" 10 "x" 80 "y" 150)) (images.Text (printf "%s · %s" site.Params.author (urls.Parse site.BaseURL).Host) (dict "font" $sans "color" $c.muted "size" 28 "x" 80 "y" 540)) -}} {{- with .label -}} {{- $filters = $filters | append (images.Text (upper .) (dict "font" $sans "color" $c.accent "size" 24 "x" 80 "y" 58)) -}} {{- end -}} {{- $out = $base | images.Filter $filters | resources.Copy (printf "og/%s.png" (md5 .page.RelPermalink)) -}} {{- end -}} {{- return $out -}}
